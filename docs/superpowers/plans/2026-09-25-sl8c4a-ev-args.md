```ctron
// tests/gui/s40_ev_args_d —— SL-8c-4a:on: 事件实参值传播域包夹具(act v2 契约)
// 蓝本 sl8c-design.md §4 值流传面(8c-3 落「求值后弃」,本片收「求值后传出」)。覆盖:
// - 值传播主判据:ev_fire 驱动 {addn(m, 5)} → ev_arg_i 回 5 → 模型 +5
//   (未实现时 act 收空表 → ev_arg_i 越界回 0 → count 恒 0,红在此)
// - 多实参按位次序:{setadj(m, 3, -1)} → a=3, b=1 位独立(count + a*10 + b)
// - 解码器契约自检:i:/b:/裸串三形 + 越界回零值
// - 旧契约零破坏:单标识符 {addn} 直发空实参不崩、d_click 按头路由不受累
use gui.{test, d_click, d_frame, d_expect_text, ev_fire, ev_arg_i, ev_arg_s, ev_arg_b, ev_arg_len}

view ArgsApp(m: Model) {
  <vbox class="root">
    <label class="title">S32 EVARGS</label>
    <label class="count">{m.count}</label>
    <button class="btn" on:click={addn(m, 5)}>ADD5</button>
    <button class="btn" on:click={setadj(m, 3, -1)}>ADJ</button>
  </vbox>
}
style root { direction: column gap: 8 padding: 16 }
style title { fg: "#ffffff" size: 24 h: 40 }
style count { fg: "#22c55e" size: 32 h: 44 }
style btn { bg: "#2563eb" fg: "#ffffff" w: 120 h: 44 }

struct Model {
    var count: I32
}

// 每事件独立处理函数(真逻辑在此;E8120 签名门按形参个数核对调用形态实参)
fn addn(m: Box[Model], n: I32) {
    m.count = m.count + n
}
fn setadj(m: Box[Model], a: I32, b: I32) {
    m.count = m.count + a * 10 + b
}

// act v2:名字 + 实参表(打标串)→ 按头路由,实参解码后转交处理函数
fn act_args(name: Str, args: List[Str], m: Box[Model]) {
    if name == "addn" {
        addn(m, ev_arg_i(args, 0))
    } else {
        if name == "setadj" {
            setadj(m, ev_arg_i(args, 0), ev_arg_i(args, 1))
        }
    }
}

// bind 通道:props 路径应答(带型 i: 前缀,8b 约定)
fn bind_all(buf: List[Str], m: Box[Model]) {
    if buf[0] == "m.count" {
        buf.push("i:" + m.count.to_string())
    }
}

// 解码器契约自检(提具名 fn,闭包体单调用配方)
fn check_decoders() {
    var xs = List[Str]()
    xs.push("i:42")
    xs.push("b:1")
    xs.push("hello")
    if ev_arg_i(xs, 0) != 42 { panic("s32: ev_arg_i 带型解码异常") }
    if ev_arg_i(xs, 9) != 0 { panic("s32: ev_arg_i 越界未回零") }
    if !ev_arg_b(xs, 1) { panic("s32: ev_arg_b 带型解码异常") }
    if ev_arg_s(xs, 2) != "hello" { panic("s32: ev_arg_s 裸串解码异常") }
    if ev_arg_len(xs) != 3 { panic("s32: ev_arg_len 异常") }
}

fn main() -> I32 {
    var src: Str = ctron_embedded()
    var m = Box[Model](Model { count: 0 })
    var rc: I32 = test(src, 320, 640,
        |buf| bind_all(buf, m),
        |name, args| act_args(name, args, m),
        |k| { },
        |t, act, key| {
            check_decoders()
            d_frame(t, |buf| bind_all(buf, m))
            d_expect_text(t, "S32 EVARGS")
            d_expect_text(t, "0")
            // d_click 按头路由(ev_fn 头 "addn" ↔ {addn(m, 5)});headless 下
            // act 由脚本直调,空实参 = 宽容契约(count 不动)
            act(d_click(t, "addn"), List[Str]())
            d_frame(t, |buf| bind_all(buf, m))
            d_expect_text(t, "0")
            // 值传播主判据:ev_fire = 真实窗口 rt_window_loop 同路径,
            // 实参求值后随名到达 act → 解码回值 → 模型变更
            ev_fire(t.t, "addn(m, 5)", |buf| bind_all(buf, m), act)
            d_frame(t, |buf| bind_all(buf, m))
            d_expect_text(t, "5")
            ev_fire(t.t, "setadj(m, 3, -1)", |buf| bind_all(buf, m), act)
            d_frame(t, |buf| bind_all(buf, m))
            d_expect_text(t, "34")
            // 旧契约零破坏:单标识符直发空实参
            ev_fire(t.t, "addn", |buf| bind_all(buf, m), act)
            d_frame(t, |buf| bind_all(buf, m))
            d_expect_text(t, "34")
        })
    return rc
}
```

- [ ] **Step 1: `ev_eval_args` 收集求值(原 1639-1680 行区域)**

签名加第 4 参 `out`,两处求值结果由弃改推入:

```ctron
// 实参受限求值:顶层逗号切分,逐段 bxv 求值后按序推入 out(打标串,8b 约定;
// 路径经 bind 通道回答)——SL-8c-4a 值不再弃,随 act 第二参交应用侧解码(ev_arg_*)
fn ev_eval_args(t: GuiTree, args: Str, bind: fn(List[Str]) -> Void, out: List[Str]) -> Void {
    var i: I32 = 0
    var start: I32 = 0
    var depth: I32 = 0
    var instr: Bool = false
    while i < args.len {
        var c: I32 = byte_at(args, i)
        if instr {
            if c == 92 {
                i += 1
            } else {
                if c == 34 {
                    instr = false
                }
            }
        } else {
            if c == 34 {
                instr = true
            } else {
                if c == 40 {
                    depth += 1
                } else {
                    if c == 41 {
                        depth -= 1
                    } else {
                        if c == 44 {
                            if depth == 0 {
                                out.push(bxv_eval(t, byte_slice(args, start, i), bind))
                                start = i + 1
                            }
                        }
                    }
                }
            }
        }
        i += 1
    }
    var last: Str = byte_slice(args, start, args.len)
    if last != "" {
        out.push(bxv_eval(t, last, bind))
    }
}
```

- [ ] **Step 2: `ev_fire` 签名加宽 + 两处 act 调用带实参表(原 1684-1724 行区域)**

```ctron
// 事件触发统一入口:单标识符直发(空实参表);调用形态求实参后按头分发,
// 实参表(打标串)随名交 act(SL-8c-4a 值传播;8c-4b auto-act 以此契约为目标)。
// 触发期形态非法 = panic 带表达式原文(编译期形态门 E8110 的运行时 belt)。
pub fn ev_fire(t: GuiTree, expr: Str, bind: fn(List[Str]) -> Void, act: fn(Str, List[Str]) -> Void) -> Void {
    var e: Str = expr
    var suf: Str = ""
    // 尾 ":数字" 实例后缀剥离(仅 ':' 领衔才认——纯数字表达式不误剥)
    var j: I32 = e.len
    while j > 0 {
        var d: I32 = byte_at(e, j - 1)
        if d >= 48 && d <= 57 {
            j -= 1
        } else {
            break
        }
    }
    if j > 0 {
        if j < e.len {
            if byte_at(e, j - 1) == 58 {
                suf = byte_slice(e, j - 1, e.len)
                e = byte_slice(e, 0, j - 1)
            }
        }
    }
    if gt_slot_simple(e) {
        act(e + suf, List[Str]())
    } else {
        var head: Str = ev_head(e)
        if head.len == 0 {
            panic("gui 事件表达式形态非法: " + expr)
        } else {
            var args = List[Str]()
            var hl: I32 = head.len
            if hl < e.len {
                if byte_at(e, hl) == 40 {
                    var close: I32 = ev_paren_end(e, hl)
                    if close < e.len {
                        ev_eval_args(t, byte_slice(e, hl + 1, close), bind, args)
                    }
                }
            }
            act(head + suf, args)
        }
    }
}
```

- [ ] **Step 3: `ev_fire` 之后(`rt_emit` 之前)插入解码器四件**

```ctron
// ==================== SL-8c-4a:事件实参解码器 ====================
// 打标串(8b 约定)→ 应用侧取回:ev_arg_s 解标展示形 / ev_arg_i 整 /
// ev_arg_b 布 / ev_arg_len 实参个数。越界回零值(宽容契约,s32 锁定);
// 裸串按字面宽容解码(gs2i 前导数字口径同样式值)。
pub fn ev_arg_len(args: List[Str]) -> I32 {
    return args.len
}

pub fn ev_arg_s(args: List[Str], i: I32) -> Str {
    if i < 0 {
        return ""
    }
    if i >= args.len {
        return ""
    }
    return bxv_show(args[i])
}

pub fn ev_arg_i(args: List[Str], i: I32) -> I32 {
    if i < 0 {
        return 0
    }
    if i >= args.len {
        return 0
    }
    var v: Str = args[i]
    if bxv_istag(v) {
        if byte_at(v, 0) == 105 {
            return bxv_iv(v)
        }
        return 0
    }
    return gs2i(v)
}

pub fn ev_arg_b(args: List[Str], i: I32) -> Bool {
    if i < 0 {
        return false
    }
    if i >= args.len {
        return false
    }
    return gt_ck_truth(args[i])
}
```
