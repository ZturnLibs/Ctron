ctc_sha() {

        shasum -a 256 "$1" | cut -d' ' -f1

```ctron
// —— D8-2 L2:工件 meta 摘要扫描(记录-比对;记录串由 seal 编排层写入,本面不重算)——
// 定位 key 在 t 中 from 起的首个出现(词边界由调用方 pkg_ctcl_pre 把守);未命中 -1
fn pkg_meta_find(t: Str, from: I32, key: Str) -> I32 {
    var i = from
    while i < t.len {
        if pkg_ctcl_at(t, i, key) {
            return i
        }
        i += 1
    }
    return 0 - 1
}

// 读 i 起(已越过键名)的 `= "值"` 形;畸形返回 ""
fn pkg_meta_strval(t: Str, i: I32) -> Str {
    var j = i
    while j < t.len && byte_at(t, j) == 32 {
        j += 1
    }
    if j < t.len && byte_at(t, j) == 61 {
        j += 1
        while j < t.len && byte_at(t, j) == 32 {
            j += 1
        }
        if j < t.len && byte_at(t, j) == 34 {
            j += 1
            var st = j
            while j < t.len && byte_at(t, j) != 34 {
                j += 1
            }
            if j < t.len {
                return byte_slice(t, st, j)
            }
        }
    }
    return ""
}

// 工件自身摘要锚(artifact 块 self_digest);meta 缺失/键缺 → ""
fn pkg_meta_self_digest(art_dir: Str) -> Str {
    var t = ""
    match read_file(art_dir + "/meta.ctcl") {
        Some(x) => {
            t = x
        }
        None => {
            return ""
        }
    }
    var i: I32 = 0
    while i < t.len {
        if pkg_ctcl_at(t, i, "self_digest") && pkg_ctcl_pre(t, i) {
            return pkg_meta_strval(t, i + 11)
        }
        i += 1
    }
    return ""
}

// 消费方工件 meta 内 dep 块 digest 需求;块/键缺 → ""(125=`}` 为块界,格式机器写入无嵌套)
fn pkg_meta_dep_digest(meta_path: Str, name: Str) -> Str {
    var t = ""
    match read_file(meta_path) {
        Some(x) => {
            t = x
        }
        None => {
            return ""
        }
    }
    var needle = "dep \"" + name + "\" \{"
    var h = pkg_meta_find(t, 0, needle)
    if h < 0 {
        return ""
    }
    var i = h + needle.len
    while i < t.len {
        if byte_at(t, i) == 125 {
            return ""
        }
        if pkg_ctcl_at(t, i, "digest") && pkg_ctcl_pre(t, i) {
            return pkg_meta_strval(t, i + 6)
        }
        i += 1
    }
    return ""
}

// 工件边校验:①被依赖工件缺 self_digest → E5055;②消费方工件边缺 dep 记录 → E5055;
// ③需求 ≠ 实际 → E5054(点名 dep 与双摘要)。顶层源消费方(ameta="")无记录面,L2 边界内跳过②。
fn pkg_art_verify(diags: List[Str], depname: Str, consumer_meta: Str, art_dir: Str) {
    var sd = pkg_meta_self_digest(art_dir)
    if sd.len == 0 {
        diag1(diags, "E5055", "", art_dir)
        return
    }
    if consumer_meta.len == 0 {
        return
    }
    var rq = pkg_meta_dep_digest(consumer_meta, depname)
    if rq.len == 0 {
        diag1(diags, "E5055", "", consumer_meta + " dep " + depname)
        return
    }
    if !seq2(rq, sd) {
        diag3(diags, "E5054", "", depname, rq, sd)
    }
}
```
